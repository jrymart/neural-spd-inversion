def plot_perf(ax):
        sns.scatterplot(
            data=sine_df,
            x="true_labels",
            y="predictions",
            s=2,
            ax=ax,
            palette="Set2"
        )
        ax.set_xlabel("Valley Spacing (m)")
        ax.set_ylabel(r"$\log_{10}(D/K)$")
