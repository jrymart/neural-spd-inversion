* Plotting Function
#+begin_src python
def plot_perf(axs):
    sns.scatterplot(
        data=sine_df,
        x='true_labels',
        y='predictions',
        s=5,
        palette="Set2",
        edgecolor=None,
        ax=axs[0]
    )
    sns.scatterplot(
        data=dd_df,
        x='drainage_density',
        y='logDoK',
        s=5,
        palette="Set2",
        edgecolor=None,
        ax=axs[1]
    )
    axs[1].ticklabel_format(style='sci', axis='x', scilimits=(0,0), useMathText=True)
    axs[0].set_xlabel("Valley Spacing (m)")
    axs[0].set_ylabel(r"$\log_{10}(D/K)$")
    axs[1].set_xlabel("Draiange Density")
    axs[1].set_ylabel("")
    axs[0].text(0,1.3, "(a)")
    axs[1].text(0,1.3, "(b)")
#+end_src
